فرم و جلوگیری از رفتار پیش‌فرض

برای هر مبحث، آموزش، مثال قابل اجرا و تمرین را باز کنید.

جلوگیری از رویداد پیش‌فرض (preventDefault)

توضیح آموزشی

preventDefault رفتار پیش‌فرض مرورگر، مثل ارسال فرم و بارگذاری مجدد، را متوقف می‌کند تا کد خودش پاسخ دهد.

فرم submit به‌طور پیش‌فرض صفحه را ارسال یا بارگذاری می‌کند. preventDefault این رفتار را متوقف می‌کند تا داده را با JavaScript بررسی کنیم. پس از اعتبارسنجی، پیام خطا را نزدیک فیلد نمایش دهید و برای دسترس‌پذیری آن را به صفحه‌خوان اعلام کنید.

کد نمونه

const form = document.createElement("form");
form.innerHTML = "<input required placeholder=نام><button>ارسال</button>";
form.addEventListener("submit", (event) => {
  event.preventDefault();
  console.log("ارسال پیش‌فرض متوقف شد");
});
document.querySelector("#demoArea").append(form);

راهنمای اجرا: برای دیدن نتیجه، دکمهٔ «اجرای مثال» را بزنید؛ خروجی زیر کد نمایش داده می‌شود.

پردازش کنترل‌های متنوع فرم
توضیح آموزشی
فرم می‌تواند ورودی متنی، فهرست کشویی و چندانتخابی، گزینه‌های رادیویی و checkbox داشته باشد؛ با FormData مقدارهای ارسال‌شده را یکجا می‌خوانیم.
به هر کنترل name بدهید تا در FormData ثبت شود. get مقدار یک فیلد و getAll همهٔ مقدارهای هم‌نام را برمی‌گرداند. مقدار فهرست کشویی را می‌توان با value تغییر داد و با select متن یک ورودی را انتخاب کرد.
کد نمونه
const area = document.querySelector("#demoArea");
area.innerHTML = `
  <form class="full-form">
    <label>نام: <input name="name" required></label>
    <button class="select-name" type="button">انتخاب متن نام</button>
    <label>شهر:
      <select name="city">
        <option value="تهران">تهران</option>
        <option value="شیراز">شیراز</option>
        <option value="تبریز">تبریز</option>
      </select>
    </label>
    <label>موضوع‌ها:
      <select name="topic" multiple size="3">
        <option value="DOM">DOM</option>
        <option value="فرم">فرم</option>
        <option value="API">API</option>
      </select>
    </label>
    <fieldset>
      <legend>سطح تجربه</legend>
      <label><input type="radio" name="level" value="مبتدی" checked> مبتدی</label>
      <label><input type="radio" name="level" value="پیشرفته"> پیشرفته</label>
    </fieldset>
    <fieldset>
      <legend>علاقه‌مندی‌ها</legend>
      <label><input type="checkbox" name="interest" value="DOM"> DOM</label>
      <label><input type="checkbox" name="interest" value="API"> API</label>
    </fieldset>
    <button type="submit">ثبت</button>
  </form>
`;
const form = area.querySelector(".full-form");
const nameInput = form.elements.name;
form.elements.city.value = "شیراز";
form.querySelector(".select-name").addEventListener("click", () => nameInput.select());
form.addEventListener("submit", (event) => {
  event.preventDefault();
  const data = new FormData(form);
  console.log("نام:", data.get("name"));
  console.log("شهر:", data.get("city"));
  console.log("موضوع‌ها:", data.getAll("topic"));
  console.log("سطح:", data.get("level"));
  console.log("علاقه‌مندی‌ها:", data.getAll("interest"));
});
راهنمای اجرا: فرم را کامل کنید، چند checkbox انتخاب کنید و دکمهٔ ثبت را بزنید.
