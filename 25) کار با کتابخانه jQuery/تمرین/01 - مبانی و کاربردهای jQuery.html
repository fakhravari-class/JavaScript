بارگذاری jQuery و انتخاب عناصر

توضیح آموزشی

jQuery کتابخانه‌ای برای انتخاب عناصر DOM و انجام کارهای رایج صفحه با API کوتاه است. در jQuery 4 می‌توان کتابخانه را از CDN رسمی بارگذاری کرد یا فایل را کنار پروژه نگه داشت.

برای CDN از نشانی code.jquery.com استفاده کنید. برای کار بدون اینترنت، نسخهٔ کامل را در پوشه‌ای مانند vendor قرار دهید و مسیر script را به ‎./vendor/jquery-4.0.0.min.js‎ تغییر دهید. تابع $ عناصر را با selector انتخاب می‌کند؛ سپس می‌توان متن، کلاس یا رویداد آن‌ها را تغییر داد.

کد نمونه

function loadJQuery() {
  if (window.jQuery) return Promise.resolve(window.jQuery);
  return new Promise((resolve, reject) => {
    const script = document.createElement("script");
    script.src = "https://code.jquery.com/jquery-4.0.0.min.js";
    script.onload = () => resolve(window.jQuery);
    script.onerror = () =>
      reject(new Error("بارگذاری jQuery ناموفق بود؛ اتصال اینترنت را بررسی کنید."));
    document.head.append(script);
  });
}
const area = document.querySelector("#demoArea");
const $ = await loadJQuery();
area.innerHTML = `
  <p class="jq-message">پیام اولیه</p>
  <button class="jq-change">تغییر پیام</button>
`;
const $area = $(area);
$area.find(".jq-message").text("پیام با jQuery تنظیم شد.").css("color", "teal");
$area.find(".jq-change").on("click", function () {
  $area.find(".jq-message").toggleClass("active").text("رویداد click با jQuery اجرا شد.");
});
console.log("jQuery:", $.fn.jquery);

راهنمای اجرا: اجرای مثال به اینترنت نیاز دارد؛ پس از بارگذاری کتابخانه، دکمهٔ داخل کادر را هم بزنید.

افکت‌های hide، toggle، fade و slide
توضیح آموزشی
jQuery افکت‌های آماده‌ای برای پنهان‌کردن، نمایان‌کردن و متحرک‌سازی عناصر دارد.
hide و show نمایش عنصر را تغییر می‌دهند؛ toggle بین این دو حالت جابه‌جا می‌شود. fadeIn و fadeOut محوشدن و ظاهرشدن را اجرا می‌کنند و slideUp و slideToggle حرکت عمودی را کنترل می‌کنند. برای جلوگیری از انباشته‌شدن افکت‌ها از stop استفاده کنید.
کد نمونه
function loadJQuery() {
  if (window.jQuery) return Promise.resolve(window.jQuery);
  return new Promise((resolve, reject) => {
    const script = document.createElement("script");
    script.src = "https://code.jquery.com/jquery-4.0.0.min.js";
    script.onload = () => resolve(window.jQuery);
    script.onerror = () =>
      reject(new Error("بارگذاری jQuery ناموفق بود؛ اتصال اینترنت را بررسی کنید."));
    document.head.append(script);
  });
}
const area = document.querySelector("#demoArea");
const $ = await loadJQuery();
area.innerHTML = `
  <div class="effect-controls">
    <button data-effect="hide">hide</button>
    <button data-effect="show">show</button>
    <button data-effect="toggle">toggle</button>
    <button data-effect="fadeOut">fadeOut</button>
    <button data-effect="fadeIn">fadeIn</button>
    <button data-effect="slideUp">slideUp</button>
    <button data-effect="slideToggle">slideToggle</button>
  </div>
  <p class="effect-panel">پنل نمونهٔ افکت‌های jQuery</p>
`;
const $area = $(area);
const $panel = $area.find(".effect-panel");
$panel.css({ background: "#e9f3ff", padding: "14px", borderRadius: "8px" });
$area.on("click", "[data-effect]", function () {
  $panel.stop(true, true);
  const effect = this.dataset.effect;
  if (effect === "hide") $panel.hide();
  if (effect === "show") $panel.show();
  if (effect === "toggle") $panel.toggle();
  if (effect === "fadeOut") $panel.fadeOut(300);
  if (effect === "fadeIn") $panel.fadeIn(300);
  if (effect === "slideUp") $panel.slideUp(300);
  if (effect === "slideToggle") $panel.slideToggle(300);
});
راهنمای اجرا: پس از اجرا، دکمه‌های مختلف را بزنید و تغییر پنل را ببینید.

رویدادهای hover و submit با jQuery
توضیح آموزشی
متد on رویداد را به عنصر متصل می‌کند و hover برای ورود و خروج نشانگر از عنصر، دو تابع اجرا می‌کند.
در فرم، رویداد submit را با on می‌گیریم و preventDefault مانع بارگذاری دوبارهٔ صفحه می‌شود. مقدار ورودی با val خوانده و خروجی با text نمایش داده می‌شود؛ برای تعامل امن، متن کاربر را با text درج کنید.
کد نمونه
function loadJQuery() {
  if (window.jQuery) return Promise.resolve(window.jQuery);
  return new Promise((resolve, reject) => {
    const script = document.createElement("script");
    script.src = "https://code.jquery.com/jquery-4.0.0.min.js";
    script.onload = () => resolve(window.jQuery);
    script.onerror = () =>
      reject(new Error("بارگذاری jQuery ناموفق بود؛ اتصال اینترنت را بررسی کنید."));
    document.head.append(script);
  });
}
const area = document.querySelector("#demoArea");
const $ = await loadJQuery();
area.innerHTML = `
  <div class="hover-box">نشانگر را روی این کادر ببرید</div>
  <form class="jq-form">
    <label>نام: <input class="jq-name" required></label>
    <button type="submit">ثبت</button>
  </form>
  <p class="jq-result" aria-live="polite"></p>
`;
const $area = $(area);
$area.find(".hover-box").css({ padding: "12px", background: "#f0f6ff" }).hover(
  function () { $(this).css("background", "#d8ebff"); },
  function () { $(this).css("background", "#f0f6ff"); },
);
$area.find(".jq-form").on("submit", function (event) {
  event.preventDefault();
  const name = $(this).find(".jq-name").val().trim();
  $area.find(".jq-result").text(`سلام ${name}`);
  console.log("فرم ارسال شد:", name);
});
راهنمای اجرا: روی کادر اشاره کنید و فرم را با یک نام ارسال کنید.
